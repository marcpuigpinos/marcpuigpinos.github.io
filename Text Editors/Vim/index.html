Vim

Install vim:

  • Debian/Ubuntu
sudo apt-get install vim
  • Fedora

To have the compiled version with clipboard support install vim-X11 package (no matter if you are using X11 or Wayland desktop).

sudo dnf install vim-X11

Now it's time to install some dependencies needed by vim-plug and the Coc pluging.

sudo dnf install git nodejs npm

Install vim-plug.

curl -fLo ~/.vim/autoload/plug.vim --create-dirs https://raw.githubusercontent.com/junegunn/vim-plug/master/plug.vim

Open vimx and type:

:PlugInstall

Note: If LSP servers are needed, it is important to install them (clangd, pyright, rust-analyzer, fortls, ...).

  • Arch
sudo pacman -S vim

Configure Vim: .vimrc​

" ============================================================================
" 1. GENERAL SETTINGS
" ============================================================================

" UI & Numbers
set number              " Show line numbers
set relativenumber     " Show relative line numbers
set ruler               " Show line and column in status line
set laststatus=2        " Always show status line
set statusline=%l,%c%V  " Custom status line (Line, Column, Percentage)
set wrap                " Enable line wrapping
set showmatch           " Highlight matching parentheses/brackets
syntax on               " Enable syntax highlighting
set mouse=a             " Enable mouse support in all modes

" Search
set incsearch           " Highlight search results as you type
set ignorecase          " Case insensitive search...
set smartcase           " ...unless a capital letter is used in the search
set hlsearch            " Highlight search results

" Indentation & Tabs
set tabstop=4           " Number of spaces per tab
set shiftwidth=4        " Number of spaces for indentation
set expandtab           " Convert tabs to spaces
set autoindent          " Copy indent from current line when starting a new one
set smartindent         " Smarter auto-indenting for code

" File & System
filetype plugin indent on " Detect file type and load plugins/indentation
set clipboard=unnamedplus " Use system clipboard for copy/paste

" Undo History
" We use expand() so Vim knows that '~' means your home directory
if !isdirectory(expand('~/.vim/undo'))
    call mkdir(expand('~/.vim/undo'), 'p')
endif
set undofile            " Maintain undo history between sessions
set undodir=~/.vim/undo
set undolevels=1000     " Maximum number of undo entries

" ============================================================================
" 2. PLUGIN MANAGEMENT (vim-plug)
" ============================================================================

call plug#begin('~/.vim/plugged')

" LSP and Autocompletion
Plug 'neoclide/coc.nvim', {'branch': 'release'}
Plug 'honza/vim-snippets'

call plug#end()

" ============================================================================
" 3. PLUGIN CONFIGURATION
" ============================================================================

" --- Coc.nvim (LSP) Configuration ---

" Use Tab to navigate completion menu
inoremap <silent><expr> <Tab> coc#pum#visible() ? coc#pum#next(1) : "\<Tab>"
inoremap <silent><expr> <S-Tab> coc#pum#visible() ? coc#pum#prev(1) : "\<S-Tab>"

" Use Enter to confirm completion
inoremap <silent><expr> <CR> coc#pum#visible() ? coc#pum#confirm() : "\<CR>"

" Automatically install language servers (LSP)
" NOTE: coc-fortran was removed to prevent 404 errors.
let g:coc_global_extensions = ['coc-clangd', 'coc-pyright', 'coc-rust-analyzer']

" ============================================================================
" 4. APPEARANCE
" ============================================================================

colorscheme desert
