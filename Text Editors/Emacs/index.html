Emacs

Install Emacs:

  • Debian/Ubuntu
sudo apt-get install emacs
  • Fedora
sudo dnf install emacs
  • Arch
sudo pacman -S emacs

Configure Emacs​

Create in the emacs configuration folder if it does not exist:

mkdir -p ~/.emacs.d

Create the following two empty files:

touch ~/.emacs.d/init.el ~/.emacs.d/emacs.custom.el

Add the content below to the init.el file.

;; Initialize package manager
(require 'package)

;; Set up package archives from MELPA and ELPA repositories
(setq package-archives
      '(("melpa" . "https://melpa.org/packages/")
        ("elpa"  . "https://elpa.gnu.org/packages/")))

;; Initialize installed packages
(package-initialize)

;; If the 'use-package' package is not installed, install it
(unless (package-installed-p 'use-package)
  (package-refresh-contents)
  (package-install 'use-package))

;; Load 'use-package' to simplify package management
(eval-when-compile
  (require 'use-package))

;; Set language environment to English and UTF-8 encoding for compatibility
(set-language-environment "English")
(prefer-coding-system 'utf-8)
(setq locale-coding-system 'utf-8)

;; Enable CUA mode for traditional copy-paste (Ctrl+C, Ctrl+V)
(cua-mode t)

;; Set default indentation to use spaces (not tabs) and set tab width to 4
(setq-default indent-tabs-mode nil)   ;; Use spaces instead of tabs
(setq-default tab-width 4)            ;; Use 4 spaces for each tab
(setq-default standard-indent 4)

;; Define a function to set up indentation settings for programming languages
(defun set-up-indent-for-all-languages ()
  (setq indent-tabs-mode nil)         ;; Use spaces
  (setq tab-width 4)                  ;; Tab width is 4 spaces
  (setq standard-indent 4))           ;; Standard indentation is 4 spaces

;; Apply the indentation setup function to all programming modes
(add-hook 'prog-mode-hook 'set-up-indent-for-all-languages)

;; Install and configure Doom themes package
(use-package doom-themes
  :ensure t
  :config
  (load-theme 'doom-dracula t))  ;; Load the Doom Dracula theme

;; Install and configure 'all-the-icons' for icon support
(use-package all-the-icons
  :ensure t)

;; Enable 'which-key' to show keybindings when pressing incomplete key sequences
(use-package which-key
  :ensure t
  :config
  (which-key-mode))

;; Enable 'company' mode for code completion
(use-package company
  :ensure t
  :config
  (add-hook 'after-init-hook 'global-company-mode))

;; Install and configure Magit for version control (Git)
(use-package magit
  :ensure t
  :bind ("C-x g" . magit-status))  ;; Bind Magit to C-x g

;; Install and configure Powerline for a better mode-line appearance
(use-package powerline
  :ensure t
  :config
  (powerline-default-theme))

;; Install and configure Ivy for completion
(use-package ivy
  :ensure t
  :config
  (ivy-mode 1))

;; Install and configure Counsel for enhanced commands and searching
(use-package counsel
  :ensure t)

;; Install and configure Swiper for an improved search experience
(use-package swiper
  :ensure t
  :bind (("C-s" . swiper)))  ;; Bind Swiper to C-s for search

;; Install and configure Smartparens for automatic pair handling (parentheses, brackets, etc.)
(use-package smartparens
  :ensure t
  :config
  (smartparens-global-mode 1))

;; Install and configure undo-fu for improved undo/redo functionality
(use-package undo-fu
  :ensure t)

;; Keybindings for undo and redo using undo-fu
(global-set-key (kbd "C-z") 'undo-fu-only-undo)
(global-set-key (kbd "C-y") 'undo-fu-only-redo)

;; Disable the startup screen and clear the initial scratch message
(setq inhibit-startup-screen t)
(setq initial-scratch-message "")

;; C/C++ indentation setup: Ensure 4-space indentation for C and C++ files
(add-hook 'c-mode-common-hook
          (lambda ()
            (setq c-basic-offset 4          ;; Set indentation to 4 spaces
                  tab-width 4               ;; Set tab width to 4 spaces
                  indent-tabs-mode nil)     ;; Use spaces, not tabs
))

(add-hook 'c++-mode-common-hook
          (lambda ()
            (setq c++-basic-offset 4          ;; Set indentation to 4 spaces
                  tab-width 4               ;; Set tab width to 4 spaces
                  indent-tabs-mode nil)     ;; Use spaces, not tabs
))

;; LSP (Language Server Protocol) configuration for C/C++
(use-package lsp-mode
  :ensure t
  :hook ((c-mode c++-mode) . lsp)  ;; Enable LSP for C/C++ modes
  :commands lsp)

(use-package lsp-ui
  :ensure t
  :commands lsp-ui-mode
  :config
  (setq lsp-ui-doc-enable t
        lsp-ui-sideline-enable t
        lsp-ui-peek-enable t))

;; Python indentation setup: Ensure 4-space indentation for Python files
(add-hook 'python-mode-hook
          (lambda ()
            (setq python-indent-offset 4
                  tab-width 4
                  indent-tabs-mode nil)))  ;; Use spaces, not tabs

;; LSP for Python with pyright
(use-package lsp-pyright
  :ensure t
  :hook (python-mode . lsp-deferred)  ;; Use lsp-deferred to load LSP asynchronously
  :config
  (setq lsp-prefer-flymake nil))  ;; Use flycheck over flymake for diagnostics

;; Additional LSP configuration for Fortran files (free-form)
(add-hook 'f90-mode-hook
          (lambda ()
            (setq indent-tabs-mode nil       ;; Use spaces
                  tab-width 4                ;; Tab width is 4
                  f90-do-indent 4            ;; Indentation after DO
                  f90-if-indent 4            ;; Indentation after IF
                  f90-type-indent 4          ;; Indentation for TYPE blocks
                  f90-program-indent 4       ;; No indent for PROGRAM
                  f90-subprogram-indent 4    ;; Indent for FUNCTION/SUBROUTINE
                  f90-begin-indent 4         ;; No indent at block start
                  f90-end-indent 4           ;; No indent at block end
                  f90-continuation-indent 4  ;; Indent for continuation lines
                  f90-indent 4)))            ;; Basic indentation level

;; LSP for Fortran with 'fortls' (Fortran Language Server)
(use-package lsp-mode
  :ensure t
  :hook ((f90-mode . lsp-deferred))
  :commands (lsp lsp-deferred)
  :config
  (setq lsp-clients-fortls-executable "fortls"
        lsp-clients-fortls-args '("--hover_signature" "--use_signature_help")))

(use-package lsp-ui
  :ensure t
  :commands lsp-ui-mode)

;; Disable UI elements that are not needed
(setq-default mode-line-format nil)
(tool-bar-mode -1)  ;; Disable the toolbar
(menu-bar-mode -1)  ;; Disable the menu bar
(scroll-bar-mode -1)  ;; Disable the scroll bar

;; Markdown
(use-package markdown-mode
  :ensure t
  :mode ("\\.md\\'" . markdown-mode)
  :config
  (setq markdown-enable-math t))
  (setq markdown-command
        "pandoc --standalone \
                --mathjax \
                --from markdown \
                --to html5")


;; Display line numbers in relative mode
(global-display-line-numbers-mode 1)
(setq display-line-numbers-type 'relative)

;; Disable backup files and auto-save files
(setq make-backup-files nil)
(setq auto-save-default nil)

;; Enable alternative file finding in Dired mode
(put 'dired-find-alternate-file 'disabled nil)

;; Auto-refresh dired on file change
(add-hook 'dired-mode-hook 'auto-revert-mode)

;; Install and configure 'multiple-cursors' for editing multiple lines
(unless (package-installed-p 'multiple-cursors)
  (package-install 'multiple-cursors))
(require 'multiple-cursors)

;; Keybindings for 'multiple-cursors' operations
(global-set-key (kbd "C-M-c") 'mc/edit-lines)
(global-set-key (kbd "C->") 'mc/mark-next-like-this)
(global-set-key (kbd "C-<") 'mc/mark-previous-like-this)
(global-set-key (kbd "C-c C-<") 'mc/mark-all-like-this)

;; Set a custom file for customizations
(setq custom-file "~/.emacs.d/emacs.custom.el")
(load-file custom-file)

;; Set the font for the Emacs interface
;;(set-face-attribute 'default nil :font "DaddyTimeMono Nerd Font" :height 160)

;; Hide the header line
(setq-default header-line-format nil)

Copilot​

If my configuration is used and the copilot section is not commentted, then it is necessary to install the following. (Requirements: to have installed npm)

To use Copilot, do the following (requirement: install npm):

Download the Emacs plugin:

git clone https://github.com/copilot-emacs/copilot.el ~/.emacs.d/copilot

Download and install copilot cli.

npm -g --prefix ~/.emacs.d/.cache/copilot install @github/copilot-language-server

Open emacs and run:

M-x copilot-install-server
M-x copilot-login

Language Server Protocol (LSP)​

To use LSP within Emacs is necessary to install the LSP servers for C/C++, Python and Fortran.

C/C++​

Install clangd.

  • Debian/Ubuntu:
sudo apt-get install clangd
  • Fedora:
sudo dnf install clang-tools-extra
  • Arch:
sudo pacman -S clang

Python​

Install pyright using pip:

pip install pyright

To install pyright system wide in some Linux distributions, it is necessary to run the following command:

pip install pyright --break-system-packages

Fortran​

Install fortls using pip:

pip install fortls

Similar to pyright, in some systems it asks to break system packages.

pip install fortls --break-system-packages

Latex math compatibility​

Install latex and pandoc.

  • Debian/Ubuntu:
sudo apt-get install texlive pandoc
  • Fedora:
sudo dnf install texlive pandoc
  • Arch:
sudo pacman -S texlive pandoc
